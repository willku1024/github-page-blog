#!/bin/bash
#
declare -i num=0
while read
do
    ((num++))
    if [ $num -ge 5 -a $num -le 10 ]; then
        echo $REPLY
    fi
done < c:\\log.txt

read  var
#等价于 sed '5,10p' /etc/passwd

#等价于 awk '{if(NR>=5 && NR<=10) {print}}' /etc/passwd
