```
#!/bin/bash
#
read -p "please input 2 number: " var1 var2

if [ $var1 -gt $var2 ];then
    let var1=$var1+$var2
    let var2=$var1-$var2
    let var1=$var1-$var2
fi

let chengji=$var1*$var2

for((k=1;0<k;k++));do
   let tmp=$var1*$k
   if [ `bc <<< "$tmp%$var2"` -eq 0 ];then
      let result=$chengji/$tmp
      echo -n "result:"
      echo $result
      exit 0
   fi
done
```
