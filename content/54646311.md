```
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int main()
{
    int i,count = 0; char * buf;
    buf = (char *)malloc(100);
    memset(buf,0,sizeof(100));
    sprintf(buf ,"%s","***H*E**L*L*O***");

    printf("初始字符串:\n%s\n",buf);
    for(i = strlen(buf)-1; i >= 0; i--)
    {
        if(buf[i] == '*')
            count++ ;
        else
        {
            buf[i + count] = buf[i];
            if(count != 0)
                buf[i] = '*';
        }
    }
    printf("处理后字符串:\n%s\n",buf);
    return 0;
}
```
