   cabinet_addtime      bigint not null default 0 comment '机柜加入系统的时间',
   cabinet_state        int not null default 0 comment '机柜状态
            0：可用
            1：不可用',
   cabinet_name         varchar(100) not null default '' comment '机柜名称，机柜的别名方便管理和查看',
   cabinet_rfid         varchar(50) not null default '' comment '机柜的唯一射频码，也是机柜的唯一标识',
   cabinet_cip          varchar(50) not null default '' comment '机柜内IP摄像头的IP地址',
   cabinet_cport        varchar(50) not null default '' comment '机柜内IP摄像头的端口号',
   cabinet_cuser        varchar(50) not null default '' comment '机柜内置摄像头用户名',
   cabinet_cpasswd      varchar(50) not null default '' comment 'IP摄像头登录密码',
   cabinet_ip           varchar(50) not null default '' comment '机柜控制主机的IP地址',
   cabinet_port         varchar(50) not null default '' comment '机柜控制主机的端口号',
   cabinet_imgpath      varchar(255) not null default '' comment '开门牌照的图片存储路径，保存相对路径，不包含主项目名称，如：
            img/upload/10/',
   cabinet_heartrate    bigint not null default 0 comment '机柜心跳包时间戳，机柜每隔30秒上传心跳包，收到心跳包则刷新此字段，若超过40秒没有收到新的心跳包则判定当前机柜断网，采集端模拟机柜生成外网异常报警信息',
   cabinet_message      text not null comment '机柜备注及描述信息',
   primary key (cabinet_id)
