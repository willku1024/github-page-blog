```
#include <iostream>

using namespace std;

int main()
{
    char sentence[50]="i am student, you are teacher.";
    char pre=' ';

    for(int i=0;i<50;i++)
    {
        if(sentence[i] != ' ' && pre == ' ')
             sentence[i]-=32;
        pre = sentence[i];

    }
    cout << sentence << endl;
    return 0;
}
```
