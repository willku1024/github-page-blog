#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct UserInfo {
  char pass[10];
  bool check;
};

int main(int argc, char *argv[]) {
  if (argc != 2) {
    printf("Invalid params\n");
    exit(1);
  }

  UserInfo info;
  info.check = false;

  memset(info.pass, 0, 10);
  strcpy(info.pass, argv[1]);

  if (0 == strcmp("shine", info.pass)) {
    info.check = true;
  }

  if (info.check) {
    printf("Check password succ!\n");
  } else {
    printf("Check password failed!\n");
  }
  return 0;
}
