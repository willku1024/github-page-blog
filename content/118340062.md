void main() {
  var a = ['红色', '蓝色', '紫色']
      .asMap()
      .map((index, element) => MapEntry(index, "$element $index"))
      .values
      .toList();

  var b = ['红色', '蓝色', '紫色'].map((element) => "$element").toList();

  print(a);
  print(b);
}
