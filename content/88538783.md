typedef  struct _t
{
    int** p;
}T;

void test(int **p)
{
    int** t = (int **)realloc(p,80000);
    p = t;
    cout << p << endl;
}
void test2(T *n)
{
    int** t = (int **)realloc(n->p,80000);
    n->p = t;
    cout << n->p << endl;
}
int main()
{
    int **p = (int **)malloc(4);
    cout << p << endl;
	test(p);
    cout << p << endl;

    T n;
    int **p2 = (int **)malloc(4);
    n.p = p2;
    //n.p = (int **)malloc(4);
    cout <<n.p << endl;
	test2(&n);
    cout << n.p << endl;
	return 0;
}
