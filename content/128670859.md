```
{
    "version": "0.2.0",
    "configurations": [
        {
            "name": "(lldb) Launch",
            "type": "cppdbg",
            "request": "launch",
            // Resolved by CMake Tools:
            "program": "${command:cmake.launchTargetPath}",
            "args": [
                //  "sdf",
                "echo-client-1", "bce33e3bb636655d41c43f4a88c67c8c", "HelloMessage", "300"
            ],
            "stopAtEntry": false,
            "cwd": "${workspaceFolder}",
            "environment": [
                {
                    // add the directory where our target was built to the PATHs
                    // it gets resolved by CMake Tools:
                    "name": "XY_ENABLE_LOG",
                    "value": "1"
                },
            ],
            // "externalConsole": false,
            // "miDebuggerPath": "/Users/xxx/.vscode/extensions/ms-vscode.cpptools-1.13.7-darwin-x64/debugAdapters/lldb-mi/bin/lldb-mi",
            "MIMode": "lldb",
            "setupCommands": [
                {
                    "description": "Enable pretty-printing for gdb",
                    "text": "-enable-pretty-printing",
                    "ignoreFailures": true
                }
            ]
        }
    ]
}

```
